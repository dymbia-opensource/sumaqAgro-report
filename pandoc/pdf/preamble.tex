% ---------------------------------------------------------------
% preamble.tex — Ajustes LaTeX para el PDF del informe
% ---------------------------------------------------------------
\usepackage{etoolbox}
\usepackage{xurl}                       % permite cortar URLs largas
\usepackage{fancyhdr}

% Numeración de páginas: número arábigo en la esquina inferior derecha de
% todas las páginas, desde la carátula (página 1), sin encabezados.
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[R]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\setlength{\headheight}{14pt}

% Texto justificado (el interlineado 1.5 se define con `linestretch` en metadata.yaml)
\emergencystretch=2em

% La carátula la genera el filtro; se desactiva el \maketitle por defecto
\renewcommand\maketitle{}

% Tablas: letra más pequeña y celdas que respetan saltos de línea
\AtBeginEnvironment{longtable}{\footnotesize\setstretch{1}}
\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{1.25}

% Encabezados #### y ##### en su propia línea (no "en línea" con el texto)
\makeatletter
\renewcommand\paragraph{\@startsection{paragraph}{4}{\z@}%
  {3ex \@plus 1ex \@minus .2ex}{1ex}{\normalfont\normalsize\bfseries}}
\renewcommand\subparagraph{\@startsection{subparagraph}{5}{\z@}%
  {2.5ex \@plus 1ex \@minus .2ex}{0.8ex}{\normalfont\normalsize\bfseries\itshape}}
\makeatother
\setcounter{secnumdepth}{-10}

% Bloques de código: cortar líneas largas
\usepackage{fvextra}
\AtBeginEnvironment{Highlighting}{\setstretch{1}}   % código con interlineado sencillo
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,breakanywhere,commandchars=\\\{\}}
\setkeys{Gin}{keepaspectratio}
